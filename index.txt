  <script>
      document.getElementById('loginForm').onsubmit = function(e) {
        e.preventDefault();
        const id = document.getElementById('id').value;
        const password = document.getElementById('password').value;

        google.script.run.withSuccessHandler(function(response) {
          if (response.valid) {
            document.getElementById('studentName').textContent = response.name;
            document.getElementById('studentClass').textContent = response.class;
            document.getElementById('studentInfo').classList.remove('hidden');
            document.getElementById('loginSection').classList.add('hidden');
            loadMonthlyData(response.id);
          } else {
            alert('ID atau password salah!');
          }
        }).validateLogin(id, password);
      };

      function loadMonthlyData(id) {
        google.script.run.withSuccessHandler(function(result) {
          const table = document.getElementById('monthlyTable');
          // Hapus baris sebelumnya kecuali header
          while (table.rows.length > 1) {
            table.deleteRow(1);
          }

          result.bulan.forEach((bulan, i) => {
            const row = table.insertRow();
            const cell1 = row.insertCell(0);
            const cell2 = row.insertCell(1);
            cell1.textContent = bulan;
            cell2.textContent = result.pembayaran[i] || '-';
          });
        }).getStudentMonthlyData(id);
      }
